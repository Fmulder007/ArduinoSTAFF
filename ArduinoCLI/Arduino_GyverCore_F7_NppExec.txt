// Arduino CLI build for GyverCore 2.0.4
// Compile only. No upload or COM port.
// Run with the main .ino file active in Notepad++.

NPP_SAVE

// Compile into the sketch directory
"C:\ArduinoCLI\arduino-cli.exe" compile --fqbn "GyverCore:avr:nano:boot=no,clock=external_16,serial=gyverUart,compiler_version=avrgcc8,timers=yes_millis,init=enable,es=enabled,bod=bod_2_7,co=disabled" --output-dir "$(CURRENT_DIRECTORY)" "$(CURRENT_DIRECTORY)"

// Stop if compilation failed
if $(EXITCODE) != 0 then
    echo BUILD FAILED!
    exit
endif

// Create LST from the newly built ELF
cmd.exe /c ""C:\Users\fm\AppData\Local\Arduino15\packages\GyverCore\hardware\avr\2.0.4\tools\avr-gcc\bin\avr-objdump.exe" -h -S -C "$(CURRENT_DIRECTORY)\$(NAME_PART).ino.elf" > "$(CURRENT_DIRECTORY)\$(NAME_PART).ino.lst""

if $(EXITCODE) != 0 then
    echo LST GENERATION FAILED!
    exit
endif

// Create SYM from the newly built ELF
cmd.exe /c ""C:\Users\fm\AppData\Local\Arduino15\packages\GyverCore\hardware\avr\2.0.4\tools\avr-gcc\bin\avr-nm.exe" -C -n -S "$(CURRENT_DIRECTORY)\$(NAME_PART).ino.elf" > "$(CURRENT_DIRECTORY)\$(NAME_PART).ino.sym""

if $(EXITCODE) != 0 then
    echo SYM GENERATION FAILED!
    exit
endif

echo BUILD SUCCESSFUL!
echo HEX, ELF, LST and SYM have been generated.
